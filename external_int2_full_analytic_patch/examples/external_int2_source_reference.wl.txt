  x2^(1 + ep)
  * (1 + x2)^ep
  * (1 + x5)^ep
  * (1 + x7)^(-1 - ep)
  * (1 + x7 + x2*x7 + r*x2*x5)^(-1 + ep);

  Exp[2*ep*EulerGamma]
  * t^(-3 - ep)
  * Gamma[1 - ep]
  * Gamma[-ep]^3
  * Gamma[ep]
  / (Gamma[-1 - 3*ep] * Gamma[-2*ep]);
